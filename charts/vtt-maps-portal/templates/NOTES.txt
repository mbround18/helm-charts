{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}

{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "vtt-maps-portal.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "vtt-maps-portal.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "vtt-maps-portal.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "vtt-maps-portal.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

2. This chart's default image (mbround18/vtt-maps-portal) is public, so no image pull
   secret is required unless you point it at a private registry.

3. DISCORD_CLIENT_SECRET and GITHUB_TOKEN are not set by default -- they come from
   external providers, so this chart cannot generate them. Create a Secret out-of-band
   and point secretEnv.<VAR>.secretName at it:
     kubectl -n {{ .Release.Namespace }} create secret generic vtt-maps-portal-secrets \
       --from-literal=DISCORD_CLIENT_SECRET=<value> --from-literal=GITHUB_TOKEN=<value> ...
   then set (e.g. via --set or a values file):
     secretEnv.DISCORD_CLIENT_SECRET.secretName=vtt-maps-portal-secrets
     secretEnv.GITHUB_TOKEN.secretName=vtt-maps-portal-secrets
   MONGO_URI/RUSTFS_ENDPOINT are wired to the bundled mongo/rustfs subcharts
   automatically; set secretEnv.MONGO_URI.secretName / secretEnv.RUSTFS_ENDPOINT.secretName
   the same way to point at an external Mongo/S3-compatible store instead.

4. JWT_SECRET_CURRENT, OAUTH_STATE_SECRET, RUSTFS_ACCESS_KEY, and RUSTFS_SECRET_KEY ARE
   generated automatically, into the "{{ .Values.generatedSecretName }}" Secret (the
   bundled rustfs subchart reads its own credentials from the same Secret). When the
   ExternalSecrets Operator is installed in this cluster (externalSecrets.mode: auto, the
   default), they're created via ESO's Password generator; otherwise this chart falls back
   to a plain Secret using the same lookup+randAlphaNum trick as the keygen chart, so the
   values stay stable across upgrades. Set generatedSecrets.enabled=false to opt out
   entirely (and add all four keys to secretEnv yourself instead).

5. The `preload.enabled` CronJob runs `mapper preload`, which clones the full maps repo
   and re-derives/uploads every image to RustFS -- expect it to take on the order of an
   hour. Size preload.schedule and preload.activeDeadlineSeconds accordingly.
